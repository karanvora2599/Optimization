cmake_minimum_required(VERSION 3.18)
project(CUDA_ResNet LANGUAGES CXX CUDA)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
set(CMAKE_CUDA_STANDARD_REQUIRED ON)

# ── CUDA Toolkit ──────────────────────────────────────────────────────────────
set(CUDA_TOOLKIT_ROOT_DIR "C:/Program Files/NVIDIA GPU Computing Toolkit/CUDA/v12.4")

# ── cuDNN — installed via:  pip install nvidia-cudnn-cu12 ─────────────────────
# Headers and DLLs live in the Python site-packages nvidia/cudnn directory.
# The import library (cudnn.lib) was generated with MinGW dlltool and is kept
# in the project libs/ folder.
set(PYTHON_SITE "C:/Users/karan/AppData/Local/Programs/Python/Python312/Lib/site-packages")
set(CUDNN_INCLUDE_DIR "${PYTHON_SITE}/nvidia/cudnn/include")
set(CUDNN_BIN_DIR     "${PYTHON_SITE}/nvidia/cudnn/bin")
set(CUDNN_LIB_DIR     "${CMAKE_SOURCE_DIR}/libs")   # contains cudnn.lib

# ── cuBLAS — headers in CUDA toolkit, .lib in CUDA toolkit ───────────────────
# The DLL ships in the CUDA toolkit bin and is already on PATH via the installer.
set(CUBLAS_INCLUDE_DIR "${CUDA_TOOLKIT_ROOT_DIR}/include")
set(CUBLAS_LIB_DIR     "${CUDA_TOOLKIT_ROOT_DIR}/lib/x64")
set(CUBLAS_BIN_DIR     "${CUDA_TOOLKIT_ROOT_DIR}/bin")

# ── Include directories ───────────────────────────────────────────────────────
include_directories(
    "${CUDA_TOOLKIT_ROOT_DIR}/include"   # cuda_runtime.h, cublas_v2.h, etc.
    "${CUDNN_INCLUDE_DIR}"               # cudnn.h
)

# ── Library search paths ──────────────────────────────────────────────────────
link_directories(
    "${CUDA_TOOLKIT_ROOT_DIR}/lib/x64"  # cuda.lib, cudart.lib, cublas.lib
    "${CUDNN_LIB_DIR}"                  # cudnn.lib (generated)
)

# ── Compiler flags ────────────────────────────────────────────────────────────
# -lineinfo       : source-level profiling info for Nsight
# --use_fast_math : fast FP; safe for training
# Architectures:  86 = RTX 30xx Ampere | 89 = RTX 40xx Ada | 80 = A100 | 75 = RTX 20xx
set(CMAKE_CUDA_FLAGS "${CMAKE_CUDA_FLAGS} -lineinfo --use_fast_math")
set(CMAKE_CUDA_ARCHITECTURES 86)   # ← Change this to match your GPU

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE  "/O2")    # MSVC release optimisation
set(CMAKE_CUDA_FLAGS_RELEASE "-O3")

# ── Sources ───────────────────────────────────────────────────────────────────
set(SOURCES
    src/main.cu
    src/kernels.cu
    src/layers.cu
    src/resnet.cu
    src/cifar10.cpp
)

add_executable(CUDA_ResNet ${SOURCES})

# ── Link libraries ────────────────────────────────────────────────────────────
target_link_libraries(CUDA_ResNet
    cuda        # cuda.lib    (CUDA driver API)
    cudart      # cudart.lib  (CUDA runtime)
    cudnn       # cudnn.lib   (from libs/ — generated by dlltool)
    cublas      # cublas.lib  (cuBLAS)
    cublasLt    # cublasLt.lib
)

# ── Post-build: copy runtime DLLs next to the .exe ────────────────────────────
# cuDNN DLLs come from the pip package; cuBLAS DLLs come from the CUDA toolkit.
set(RUNTIME_DLLS
    "${CUDNN_BIN_DIR}/cudnn64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_ops64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_cnn64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_adv64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_graph64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_engines_precompiled64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_engines_runtime_compiled64_9.dll"
    "${CUDNN_BIN_DIR}/cudnn_heuristic64_9.dll"
    "${CUBLAS_BIN_DIR}/cublas64_12.dll"
    "${CUBLAS_BIN_DIR}/cublasLt64_12.dll"
)

foreach(dll ${RUNTIME_DLLS})
    if(EXISTS "${dll}")
        add_custom_command(TARGET CUDA_ResNet POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "${dll}" "$<TARGET_FILE_DIR:CUDA_ResNet>"
            COMMENT "Copying ${dll}")
    endif()
endforeach()

message(STATUS "Build type    : ${CMAKE_BUILD_TYPE}")
message(STATUS "CUDA arch     : ${CMAKE_CUDA_ARCHITECTURES}")
message(STATUS "CUDA toolkit  : ${CUDA_TOOLKIT_ROOT_DIR}")
message(STATUS "cuDNN headers : ${CUDNN_INCLUDE_DIR}")
message(STATUS "cuDNN lib     : ${CUDNN_LIB_DIR}/cudnn.lib")
